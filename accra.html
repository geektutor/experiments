What is Google Apps Script?

Google Apps Script is a cloud-based JavaScript runtime powered by Chrome's V8 engine running on Google's global infrastructure. It lets you automate Google Workspace (Sheets, Docs, Drive) and turn any spreadsheet into a completely free, serverless REST API using standard doGet(e) and doPost(e) endpoints.

Step 3 of 6 • Est. 8 minutes

Google Apps Script Backend (Code.gs)

In your Google Sheet, navigate to Extensions > Apps Script. Replace everything in Code.gs with this code. It exposes doPost to record scores and doGet to return a leaderboard sorted descending by score.

Code.gs (Apps Script Backend) 
const SHEET_NAME = "Sheet1";

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
    
    // If headers don't exist yet, add them
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Name", "Score"]);
    }
    
    // Append the player's name and score
    sheet.appendRow([new Date(), data.name, data.stoppedAt]);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const rows = sheet.getDataRange().getValues();
  
  let scores = [];
  // Skip header row (start at i = 1)
  for (let i = 1; i < rows.length; i++) {
    scores.push({
      timestamp: rows[i][0],
      name: rows[i][1],
      stoppedAt: rows[i][2] // This holds the final score
    });
  }
  
  // Sort descending by score (highest score first)
  scores.sort((a, b) => b.stoppedAt - a.stoppedAt);
  
  // Keep top 10 unique top scores or return all
  return ContentService.createTextOutput(JSON.stringify(scores))
    .setMimeType(ContentService.MimeType.JSON);
}

Deployment Steps (Crucial: Select "Anyone")

  1. In the Apps Script editor, click Deploy > New deployment.
  2. Click the gear icon next to "Select type" and choose Web app.
  3. Description: Stroop Game API
  4. Execute as: Me (your Google email).
  5. Who has access: Anyone. (Selecting "Only myself" will trigger CORS / auth errors in your browser game).
  6. Click Deploy, grant required spreadsheet permissions, and copy the Web App URL!

    <script>
        // PASTE YOUR DEPLOYED GOOGLE APPS SCRIPT WEB APP URL HERE:
        const SCRIPT_URL = 'YOUR_APPS_SCRIPT_WEB_APP_URL';

        let name = "";
        let score = 0;
        let streak = 0;
        let roundTime = 5000; // 5 seconds per round
        let timerInterval = null;
        let startTime = null;
        let correctCardIndex = -1;

        const COLORS = [
            { name: 'RED', hex: '#f85149', cssName: 'red' },
            { name: 'BLUE', hex: '#58a6ff', cssName: 'blue' },
            { name: 'GREEN', hex: '#3fb950', cssName: 'green' },
            { name: 'YELLOW', hex: '#d29922', cssName: 'yellow' }
        ];

        function startGame() {
            name = document.getElementById('username').value.trim();
            if (!name) return alert("Identify yourself, dev.");
            score = 0;
            streak = 0;
            switchScreen('screen-game');
            nextRound();
        }

        function nextRound() {
            document.getElementById('current-score').innerText = score;
            document.getElementById('current-streak').innerText = streak;

            // Pick a rule type: 0 = Background color, 1 = Text color, 2 = Text meaning word
            const ruleType = Math.floor(Math.random() * 3);
            
            // Pick target color object
            const targetColor = COLORS[Math.floor(Math.random() * COLORS.length)];
            
            if (ruleType === 0) {
                document.getElementById('instruction').innerText = `CLICK FRAME WITH BACKGROUND: ${targetColor.name}`;
            } else if (ruleType === 1) {
                document.getElementById('instruction').innerText = `CLICK FRAME WITH TEXT COLOR: ${targetColor.name}`;
            } else {
                document.getElementById('instruction').innerText = `CLICK FRAME WITH WORD: ${targetColor.name}`;
            }

            // Generate properties for 4 cards ensuring one is correct and others are distractors
            correctCardIndex = Math.floor(Math.random() * 4);
            let cardData = [];

            for (let i = 0; i < 4; i++) {
                if (i === correctCardIndex) {
                    if (ruleType === 0) {
                        cardData.push({
                            bg: targetColor.hex,
                            text: COLORS[Math.floor(Math.random() * COLORS.length)].name,
                            textColor: COLORS[Math.floor(Math.random() * COLORS.length)].hex
                        });
                    } else if (ruleType === 1) {
                        cardData.push({
                            bg: COLORS[Math.floor(Math.random() * COLORS.length)].hex,
                            text: COLORS[Math.floor(Math.random() * COLORS.length)].name,
                            textColor: targetColor.hex
                        });
                    } else {
                        cardData.push({
                            bg: COLORS[Math.floor(Math.random() * COLORS.length)].hex,
                            text: targetColor.name,
                            textColor: COLORS[Math.floor(Math.random() * COLORS.length)].hex
                        });
                    }
                } else {
                    let dBg, dText, dTextCol;
                    do {
                        dBg = COLORS[Math.floor(Math.random() * COLORS.length)].hex;
                        dText = COLORS[Math.floor(Math.random() * COLORS.length)].name;
                        dTextCol = COLORS[Math.floor(Math.random() * COLORS.length)].hex;
                    } while (
                        (ruleType === 0 && dBg === targetColor.hex) ||
                        (ruleType === 1 && dTextCol === targetColor.hex) ||
                        (ruleType === 2 && dText === targetColor.name)
                    );
                    cardData.push({ bg: dBg, text: dText, textColor: dTextCol });
                }
            }

            // Render cards to UI
            for (let i = 0; i < 4; i++) {
                const cardEl = document.getElementById(`card-${i}`);
                cardEl.style.backgroundColor = cardData[i].bg;
                cardEl.style.color = cardData[i].textColor;
                cardEl.innerText = cardData[i].text;
            }

            startTimer();
        }

        function startTimer() {
            clearInterval(timerInterval);
            startTime = Date.now();
            const bar = document.getElementById('timerBar');
            bar.style.width = '100%';

            timerInterval = setInterval(() => {
                const elapsed = Date.now() - startTime;
                const remaining = roundTime - elapsed;
                const pct = Math.max(0, (remaining / roundTime) * 100);
                bar.style.width = pct + '%';

                if (remaining <= 0) {
                    clearInterval(timerInterval);
                    gameOver();
                }
            }, 50);
        }

        function handleCardClick(index) {
            clearInterval(timerInterval);
            if (index === correctCardIndex) {
                score += 100 + (streak * 10);
                streak++;
                if (roundTime > 2500) roundTime -= 150; 
                nextRound();
            } else {
                gameOver();
            }
        }

        function gameOver() {
            clearInterval(timerInterval);
            switchScreen('screen-result');
            document.getElementById('final-score').innerText = score;

            const title = document.getElementById('result-title');
            if (score > 1500) title.innerText = "AI OVERLORD";
            else if (score > 800) title.innerText = "STAFF ENGINEER";
            else if (score > 300) title.innerText = "JUNIOR DEV";
            else title.innerText = "INTERN BUG";

            saveScore(score);
        }

        function saveScore(finalScore) {
            fetch(SCRIPT_URL, {
                method: 'POST',
                body: JSON.stringify({ name: name, stoppedAt: finalScore, diff: "Score" })
            })
            .then(() => fetchLeaderboard())
            .catch(err => console.error(err));
        }

        function fetchLeaderboard() {
            fetch(SCRIPT_URL)
            .then(res => res.json())
            .then(data => {
                const list = document.getElementById('leaderboard-list');
                let html = '';
                data.sort((a, b) => b.stoppedAt - a.stoppedAt);
                data.slice(0, 5).forEach(p => {
                    html += `<li><span>${escapeHtml(p.name)}</span> <strong>${p.stoppedAt} pts</strong></li>`;
                });
                list.innerHTML = html;
            }).catch(() => {
                document.getElementById('leaderboard-list').innerHTML = `<li style="justify-content:center;">Score saved locally!</li>`;
            });
        }

        function resetGame() {
            roundTime = 5000;
            switchScreen('screen-login');
        }

        function switchScreen(id) {
            document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
            document.getElementById(id).classList.add('active');
        }

        function escapeHtml(text) {
            return text.replace(/[&<>"']/g, function(m) {
                return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }[m];
            });
        }
    </script>
</body>

Troubleshooting Tips

Sheet tab name: Verify the bottom tab in your sheet is called Sheet1, which matches const SHEET_NAME = "Sheet1";.
